package engine

// Base probe for ledger F11 (round 4): does the class-claims half-propagation
// (container claim synced, pod-level resourceClaims never synced) reproduce on
// the merge base e0fc4c18? Two shapes:
//
//   - E1:  template carries ONLY the claim pair (no requests/limits).
//   - E1b: template pairs the claim with requests/limits (the natural shape).
//
// Both drive syncRuntimeSpec exactly like the round-4 harness does.

import (
	"context"
	"testing"

	workloadv1alpha1 "github.com/fluid-cloudnative/advanced-statefulset/api/workload/v1alpha1"
	datav1alpha1 "github.com/fluid-cloudnative/fluid/api/v1alpha1"
	cruntime "github.com/fluid-cloudnative/fluid/pkg/runtime"
	corev1 "k8s.io/api/core/v1"
	"k8s.io/apimachinery/pkg/api/resource"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/types"
	ctrl "sigs.k8s.io/controller-runtime"
	"sigs.k8s.io/controller-runtime/pkg/client/fake"
)

func v6177r4BaseEngine(t *testing.T, asts *workloadv1alpha1.AdvancedStatefulSet,
	runtime *datav1alpha1.CacheRuntime, runtimeClass *datav1alpha1.CacheRuntimeClass) *CacheEngine {
	client := fake.NewClientBuilder().
		WithScheme(CacheEngineTestScheme).
		WithObjects(asts, runtime, runtimeClass).
		Build()
	return &CacheEngine{
		name:      "test",
		namespace: "default",
		Client:    client,
		Log:       ctrl.Log.WithName("verify-base-6177"),
	}
}

func v6177r4BaseRun(t *testing.T, templateResources corev1.ResourceRequirements) {
	replicas := int32(1)
	podTemplate := func(res corev1.ResourceRequirements) corev1.PodTemplateSpec {
		return corev1.PodTemplateSpec{
			Spec: corev1.PodSpec{
				ResourceClaims: []corev1.PodResourceClaim{{Name: "shm"}},
				Containers: []corev1.Container{{
					Name:      "worker",
					Image:     "fluid/cache:v1",
					Resources: *res.DeepCopy(),
				}},
			},
		}
	}
	asts := &workloadv1alpha1.AdvancedStatefulSet{
		ObjectMeta: metav1.ObjectMeta{Name: "test-worker", Namespace: "default"},
		Spec: workloadv1alpha1.AdvancedStatefulSetSpec{
			Replicas: &replicas,
			Template: func() corev1.PodTemplateSpec {
				pt := podTemplate(corev1.ResourceRequirements{})
				pt.Spec.ResourceClaims = nil // workload created before the class edit
				return pt
			}(),
		},
	}
	runtimeObj := &datav1alpha1.CacheRuntime{
		ObjectMeta: metav1.ObjectMeta{Name: "test", Namespace: "default"},
		Spec: datav1alpha1.CacheRuntimeSpec{
			RuntimeClassName: "test-class",
			Worker:           datav1alpha1.CacheRuntimeWorkerSpec{Replicas: 1},
		},
	}
	runtimeClass := &datav1alpha1.CacheRuntimeClass{
		ObjectMeta: metav1.ObjectMeta{Name: "test-class"},
		Topology: &datav1alpha1.RuntimeTopology{
			Worker: &datav1alpha1.RuntimeComponentDefinition{Template: podTemplate(templateResources)},
		},
	}
	engine := v6177r4BaseEngine(t, asts, runtimeObj, runtimeClass)

	if err := engine.syncRuntimeSpec(cruntime.ReconcileRequestContext{Context: context.TODO()}, runtimeObj, runtimeClass); err != nil {
		t.Fatalf("syncRuntimeSpec failed: %v", err)
	}

	key := types.NamespacedName{Name: "test-worker", Namespace: "default"}
	synced := &workloadv1alpha1.AdvancedStatefulSet{}
	if err := engine.Client.Get(context.TODO(), key, synced); err != nil {
		t.Fatalf("get ASTS: %v", err)
	}

	container := synced.Spec.Template.Spec.Containers[0]
	declared := map[string]bool{}
	for _, pc := range synced.Spec.Template.Spec.ResourceClaims {
		declared[pc.Name] = true
	}
	for _, claim := range container.Resources.Claims {
		if !declared[claim.Name] {
			t.Errorf("F11 shape: workload container references claim %q not declared at pod level (%v) — the wedge reproduces on this ref",
				claim.Name, synced.Spec.Template.Spec.ResourceClaims)
		}
	}
}

// E1 at base: template carries ONLY the claim pair.
func TestE1BaseClaimOnlyTemplate(t *testing.T) {
	v6177r4BaseRun(t, corev1.ResourceRequirements{Claims: []corev1.ResourceClaim{{Name: "shm"}}})
}

// E1b at base: template pairs the claim with requests/limits.
func TestE1bBasePairedTemplate(t *testing.T) {
	v6177r4BaseRun(t, corev1.ResourceRequirements{
		Requests: corev1.ResourceList{corev1.ResourceCPU: resource.MustParse("1")},
		Claims:   []corev1.ResourceClaim{{Name: "shm"}},
	})
}
