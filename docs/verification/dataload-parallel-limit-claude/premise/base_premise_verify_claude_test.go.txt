package v1alpha1

// P0 premise check for https://github.com/fluid-cloudnative/fluid/pull/5754,
// run against the BASE branch (82e490e4) WITHOUT the patch.
//
// PR claim (quoted from the PR body): "Previously, SetDataOperationInProgress
// would always append new operation names to a comma-separated string,
// effectively allowing multiple tasks to run on the same dataset
// simultaneously."
// Linked issue #1138 (quoted): "what would happen if multiple same DataLoad
// task start running?"
//
// This test documents the BASE behavior: a second same-type operation
// registers unconditionally, with no limit check anywhere on the path.
// PASS here == premise confirmed (the problem exists on base).

import "testing"

func TestVerifyBasePremise_SecondOperationRegistersUnconditionally(t *testing.T) {
	dataset := &Dataset{
		Status: DatasetStatus{
			OperationRef: map[string]string{"DataLoad": "load-1"},
		},
	}

	dataset.SetDataOperationInProgress("DataLoad", "load-2")
	got := dataset.Status.OperationRef["DataLoad"]
	if got != "load-1,load-2" {
		t.Fatalf("on base, second registration result = %q, want %q (unconditional append)", got, "load-1,load-2")
	}
}
